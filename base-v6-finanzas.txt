===== PAGO ENTITY =====
package pe.vichaval.gestionriego.tesoreria.entity;

import jakarta.persistence.*;

import java.math.*;
import java.time.*;

import pe.vichaval.gestionriego.agricultor.entity.Agricultor;
import pe.vichaval.gestionriego.tesoreria.dto.PagoForm;

@Entity
@Table(name = "pagos")
public class Pago {
    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;
    @Column(nullable = false, unique = true, length = 30)
    private String codigo;
    @Column(nullable = false)
    private BigDecimal monto;
    @Column(nullable = false)
    private LocalDate fecha;
    @Column(nullable = false, length = 150)
    private String concepto;
    @Column(nullable = false, unique = true, length = 50)
    private String comprobante;
    @Column(nullable = false, length = 30)
    private String metodo;
    @Column(nullable = false, length = 30)
    private String estado;
    @Column(name = "motivo_rechazo", length = 500)
    private String motivoRechazo;
    @Column(name = "motivo_anulacion", length = 500)
    private String motivoAnulacion;
    @Column(name = "fecha_creacion", nullable = false)
    private OffsetDateTime fechaCreacion = OffsetDateTime.now();
    @Column(name = "fecha_actualizacion", nullable = false)
    private OffsetDateTime fechaActualizacion = OffsetDateTime.now();
    @ManyToOne(fetch = FetchType.LAZY, optional = false)
    @JoinColumn(name = "agricultor_id", nullable = false)
    private Agricultor agricultor;

    protected Pago() {
    }

    public Pago(String c, PagoForm f, Agricultor a) {
        codigo = c;
        estado = "EFECTIVO".equals(f.getMetodo()) ? "CONFIRMADO" : "PENDIENTE";
        cargar(f, a);
    }

    private void cargar(PagoForm f, Agricultor a) {
        monto = f.getMonto();
        fecha = f.getFecha();
        concepto = f.getConcepto();
        comprobante = f.getComprobante();
        metodo = f.getMetodo();
        agricultor = a;
        fechaActualizacion = OffsetDateTime.now();
    }

    public void confirmar() {
        if (!"PENDIENTE".equals(estado)) throw new IllegalStateException("Solo pagos pendientes");
        estado = "CONFIRMADO";
    }

    public void rechazar(String m) {
        if (!"PENDIENTE".equals(estado)) throw new IllegalStateException("Solo pagos pendientes");
        if (m == null || m.isBlank()) throw new IllegalArgumentException("Motivo obligatorio");
        estado = "RECHAZADO";
        motivoRechazo = m;
    }

    public void anular(String m) {
        if (!"CONFIRMADO".equals(estado)) throw new IllegalStateException("Solo pagos confirmados");
        if (m == null || m.isBlank()) throw new IllegalArgumentException("Motivo obligatorio");
        estado = "ANULADO";
        motivoAnulacion = m;
    }

    public Long getId() {
        return id;
    }

    public String getCodigo() {
        return codigo;
    }

    public BigDecimal getMonto() {
        return monto;
    }

    public LocalDate getFecha() {
        return fecha;
    }

    public String getConcepto() {
        return concepto;
    }

    public String getComprobante() {
        return comprobante;
    }

    public String getMetodo() {
        return metodo;
    }

    public String getEstado() {
        return estado;
    }

    public Agricultor getAgricultor() {
        return agricultor;
    }
}

package pe.vichaval.gestionriego.tesoreria.repository;

import java.math.BigDecimal;
import java.util.List;

import org.springframework.data.jpa.repository.*;
import pe.vichaval.gestionriego.tesoreria.entity.Pago;

public interface PagoRepository extends JpaRepository<Pago, Long> {
    boolean existsByComprobante(String c);

    long countByAgricultorId(Long id);

    @Query("select coalesce(sum(p.monto),0) from Pago p where p.estado='CONFIRMADO'")
    BigDecimal total();

    long countByEstado(String e);

    @Override
    @EntityGraph(attributePaths = "agricultor")
    List<Pago> findAll();
}

package pe.vichaval.gestionriego.tesoreria.service;

import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import pe.vichaval.gestionriego.agricultor.repository.AgricultorRepository;
import pe.vichaval.gestionriego.tesoreria.dto.PagoForm;
import pe.vichaval.gestionriego.tesoreria.entity.Pago;
import pe.vichaval.gestionriego.tesoreria.repository.PagoRepository;

@Service
@Transactional
public class PagoService {
    private final PagoRepository p;
    private final AgricultorRepository a;

    public PagoService(PagoRepository p, AgricultorRepository a) {
        this.p = p;
        this.a = a;
    }

    public Pago get(Long id) {
        return p.findById(id).orElseThrow(() -> new IllegalArgumentException("Pago no encontrado"));
    }

    public void registrar(PagoForm f) {
        if (p.existsByComprobante(f.getComprobante())) throw new IllegalArgumentException("Comprobante duplicado");
        var ag = a.findById(f.getAgricultorId()).orElseThrow();
        p.save(new Pago("PG-%06d".formatted(p.count() + 1), f, ag));
    }

    public void confirmar(Long id) {
        get(id).confirmar();
    }

    public void rechazar(Long id, String m) {
        get(id).rechazar(m);
    }

    public void anular(Long id, String m) {
        get(id).anular(m);
    }
}

package pe.vichaval.gestionriego.tesoreria.controller;

import jakarta.validation.Valid;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.validation.BindingResult;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.ModelAttribute;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.servlet.mvc.support.RedirectAttributes;
import pe.vichaval.gestionriego.agricultor.repository.AgricultorRepository;
import pe.vichaval.gestionriego.tesoreria.dto.PagoForm;
import pe.vichaval.gestionriego.tesoreria.repository.PagoRepository;
import pe.vichaval.gestionriego.tesoreria.service.PagoService;

@Controller
@RequestMapping("/tesoreria")
public class TesoreriaWebController {

    private final PagoRepository pagoRepository;
    private final AgricultorRepository agricultorRepository;
    private final PagoService pagoService;

    public TesoreriaWebController(
            PagoRepository pagoRepository,
            AgricultorRepository agricultorRepository,
            PagoService pagoService
    ) {
        this.pagoRepository = pagoRepository;
        this.agricultorRepository = agricultorRepository;
        this.pagoService = pagoService;
    }

    @GetMapping
    public String mostrarTesoreria(Model model) {

        cargarDatos(model);

        if (!model.containsAttribute("form")) {
            model.addAttribute(
                    "form",
                    new PagoForm()
            );
        }

        return "tesoreria/index";
    }

    @PostMapping("/pagos/guardar")
    public String registrarPago(
            @Valid @ModelAttribute("form") PagoForm form,
            BindingResult bindingResult,
            Model model,
            RedirectAttributes redirectAttributes
    ) {
        /*
         * Si existe algún error de validación, se vuelve a mostrar
         * el formulario conservando los valores ingresados.
         */
        if (bindingResult.hasErrors()) {
            cargarDatos(model);
            return "tesoreria/index";
        }

        try {
            pagoService.registrar(form);

            redirectAttributes.addFlashAttribute(
                    "exito",
                    "Pago registrado correctamente."
            );

            return "redirect:/tesoreria";

        } catch (RuntimeException exception) {

            bindingResult.reject(
                    "errorNegocio",
                    exception.getMessage()
            );

            cargarDatos(model);

            return "tesoreria/index";
        }
    }

    @PostMapping("/pagos/{id}/confirmar")
    public String confirmarPago(
            @PathVariable Long id,
            RedirectAttributes redirectAttributes
    ) {
        return ejecutarOperacion(
                () -> pagoService.confirmar(id),
                redirectAttributes,
                "Pago confirmado correctamente."
        );
    }

    @PostMapping("/pagos/{id}/rechazar")
    public String rechazarPago(
            @PathVariable Long id,
            @RequestParam String motivo,
            RedirectAttributes redirectAttributes
    ) {
        return ejecutarOperacion(
                () -> pagoService.rechazar(id, motivo),
                redirectAttributes,
                "Pago rechazado correctamente."
        );
    }

    @PostMapping("/pagos/{id}/anular")
    public String anularPago(
            @PathVariable Long id,
            @RequestParam String motivo,
            RedirectAttributes redirectAttributes
    ) {
        return ejecutarOperacion(
                () -> pagoService.anular(id, motivo),
                redirectAttributes,
                "Pago anulado correctamente."
        );
    }

    private String ejecutarOperacion(
            Runnable operacion,
            RedirectAttributes redirectAttributes,
            String mensajeExito
    ) {
        try {
            operacion.run();

            redirectAttributes.addFlashAttribute(
                    "exito",
                    mensajeExito
            );

        } catch (RuntimeException exception) {

            redirectAttributes.addFlashAttribute(
                    "error",
                    exception.getMessage()
            );
        }

        return "redirect:/tesoreria";
    }

    private void cargarDatos(Model model) {

        model.addAttribute(
                "pagos",
                pagoRepository.findAll()
        );

        model.addAttribute(
                "total",
                pagoRepository.total()
        );

        model.addAttribute(
                "agricultores",
                agricultorRepository
                        .findAll()
                        .stream()
                        .filter(agricultor ->
                                "ACTIVO".equals(
                                        agricultor.getEstado()
                                )
                        )
                        .toList()
        );
    }
}

package pe.vichaval.gestionriego.dashboard.service;

import java.math.BigDecimal;
import java.time.LocalDate;

import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import pe.vichaval.gestionriego.agricultor.repository.AgricultorRepository;
import pe.vichaval.gestionriego.dashboard.dto.DashboardResumen;
import pe.vichaval.gestionriego.parcela.repository.ParcelaRepository;
import pe.vichaval.gestionriego.riego.repository.TurnoRiegoRepository;
import pe.vichaval.gestionriego.tesoreria.repository.PagoRepository;

@Service
public class DashboardService {

    private final AgricultorRepository agricultorRepository;
    private final ParcelaRepository parcelaRepository;
    private final TurnoRiegoRepository turnoRepository;
    private final PagoRepository pagoRepository;

    public DashboardService(
            AgricultorRepository agricultorRepository,
            ParcelaRepository parcelaRepository,
            TurnoRiegoRepository turnoRepository,
            PagoRepository pagoRepository
    ) {
        this.agricultorRepository = agricultorRepository;
        this.parcelaRepository = parcelaRepository;
        this.turnoRepository = turnoRepository;
        this.pagoRepository = pagoRepository;
    }

    @Transactional(readOnly = true)
    public DashboardResumen obtenerResumen() {

        LocalDate fechaActual = LocalDate.now();

        long turnosHoy =
                turnoRepository.countByFecha(fechaActual);

        long turnosProgramados =
                turnoRepository.countByFechaAndEstado(
                        fechaActual,
                        "PROGRAMADO"
                );

        long agricultoresActivos =
                agricultorRepository.countByEstado("ACTIVO");

        long parcelasRegistradas =
                parcelaRepository.count();

        BigDecimal totalIngresos =
                pagoRepository.total();

        long cantidadPagos =
                pagoRepository.count();

        /*
         * Estos valores serán sustituidos cuando se implemente
         * la entidad CuentaPorCobrar o Deuda.
         */
        BigDecimal totalDeuda =
                BigDecimal.ZERO;

        long agricultoresConDeuda =
                0L;

        return new DashboardResumen(
                turnosHoy,
                turnosProgramados,
                agricultoresActivos,
                parcelasRegistradas,
                totalIngresos,
                cantidadPagos,
                totalDeuda,
                agricultoresConDeuda
        );
    }
}

package pe.vichaval.gestionriego.config;

import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.security.config.annotation.method.configuration.EnableMethodSecurity;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.crypto.factory.PasswordEncoderFactories;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.security.web.SecurityFilterChain;

@Configuration
@EnableMethodSecurity
public class SecurityConfig {

    @Bean
    public SecurityFilterChain securityFilterChain(
            HttpSecurity http
    ) throws Exception {

        http.authorizeHttpRequests(authorize -> authorize

                // Recursos públicos
                .requestMatchers(
                        "/login",
                        "/error",
                        "/error/**",
                        "/css/**",
                        "/js/**",
                        "/images/**",
                        "/icons/**",
                        "/favicon.ico",
                        "/actuator/health"
                ).permitAll()

                // Acceso compartido al dashboard
                .requestMatchers(
                        "/",
                        "/dashboard"
                ).hasAnyRole(
                        "ADMINISTRADOR",
                        "OPERADOR",
                        "TESORERIA"
                )

                // Gestión operativa
                .requestMatchers(
                        "/agricultores/**",
                        "/parcelas/**",
                        "/turnos/**"
                ).hasAnyRole(
                        "ADMINISTRADOR",
                        "OPERADOR"
                )

                // Gestión económica
                .requestMatchers(
                        "/tesoreria/**"
                ).hasAnyRole(
                        "ADMINISTRADOR",
                        "TESORERIA"
                )

                // Reportes financieros
                .requestMatchers(
                        "/reportes/financieros/**"
                ).hasAnyRole(
                        "ADMINISTRADOR",
                        "TESORERIA"
                )

                // Reportes operativos e inteligencia
                .requestMatchers(
                        "/reportes/**",
                        "/inteligencia/**"
                ).hasAnyRole(
                        "ADMINISTRADOR",
                        "OPERADOR"
                )

                // Administración de cuentas y roles
                .requestMatchers(
                        "/usuarios/**",
                        "/roles/**"
                ).hasRole(
                        "ADMINISTRADOR"
                )

                // Cualquier otra ruta exige autenticación
                .anyRequest()
                .authenticated()
        );

        http.formLogin(form -> form
                .loginPage("/login")
                .loginProcessingUrl("/login")
                .defaultSuccessUrl("/dashboard", true)
                .failureUrl("/login?error")
                .permitAll()
        );

        http.logout(logout -> logout
                .logoutUrl("/logout")
                .logoutSuccessUrl("/login?logout")
                .invalidateHttpSession(true)
                .clearAuthentication(true)
                .deleteCookies("VICHAYAL_SESSION")
                .permitAll()
        );

        http.exceptionHandling(exception -> exception
                .accessDeniedPage("/error/403")
        );

        return http.build();
    }

    @Bean
    public PasswordEncoder passwordEncoder() {
        return PasswordEncoderFactories
                .createDelegatingPasswordEncoder();
    }
}
