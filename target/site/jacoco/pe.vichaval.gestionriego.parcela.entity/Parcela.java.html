package pe.vichaval.gestionriego.parcela.entity;
import jakarta.persistence.*; import java.math.BigDecimal; import pe.vichaval.gestionriego.agricultor.entity.Agricultor; import pe.vichaval.gestionriego.parcela.dto.ParcelaForm;
@Entity @Table(name="parcelas") public class Parcela { @Id @GeneratedValue(strategy=GenerationType.IDENTITY) private Long id; @Column(nullable=false,unique=true,length=30) private String codigo; @Column(nullable=false,length=100) private String nombre; @Column(nullable=false) private BigDecimal hectareas; @Column(length=100) private String cultivo; @Column(nullable=false,length=30) private String estado="ACTIVA"; @ManyToOne(fetch=FetchType.LAZY,optional=false) @JoinColumn(name="agricultor_id",nullable=false) private Agricultor agricultor; protected Parcela(){} public Parcela(ParcelaForm f,Agricultor a){actualizar(f,a);} public void actualizar(ParcelaForm f,Agricultor a){codigo=f.getCodigo().trim().toUpperCase();nombre=f.getNombre().trim();hectareas=f.getHectareas();cultivo=f.getCultivo()==null?null:f.getCultivo().trim();estado=f.getEstado();agricultor=a;} public Long getId(){return id;} public String getCodigo(){return codigo;}public String getNombre(){return nombre;}public BigDecimal getHectareas(){return hectareas;}public String getCultivo(){return cultivo;}public String getEstado(){return estado;}public Agricultor getAgricultor(){return agricultor;} }
