package pe.vichaval.gestionriego.config;

import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.security.config.annotation.method.configuration.EnableMethodSecurity;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.crypto.factory.PasswordEncoderFactories;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.security.web.SecurityFilterChain;

@Configuration
@EnableMethodSecurity
public class SecurityConfig {

    @Bean
    public SecurityFilterChain securityFilterChain(
            HttpSecurity http
    ) throws Exception {

        http.authorizeHttpRequests(authorize -> authorize

                // Recursos públicos
                .requestMatchers(
                        "/login",
                        "/error",
                        "/error/**",
                        "/css/**",
                        "/js/**",
                        "/images/**",
                        "/icons/**",
                        "/favicon.ico",
                        "/actuator/health"
                ).permitAll()

                // Acceso compartido al dashboard
                .requestMatchers(
                        "/",
                        "/dashboard"
                ).hasAnyRole(
                        "ADMINISTRADOR",
                        "OPERADOR",
                        "TESORERIA"
                )

                // Gestión operativa
                .requestMatchers(
                        "/agricultores/**",
                        "/parcelas/**",
                        "/turnos/**"
                ).hasAnyRole(
                        "ADMINISTRADOR",
                        "OPERADOR"
                )

                // Gestión económica
                .requestMatchers(
                        "/tesoreria/**"
                ).hasAnyRole(
                        "ADMINISTRADOR",
                        "TESORERIA"
                )

                // Reportes financieros
                .requestMatchers(
                        "/reportes/financieros/**"
                ).hasAnyRole(
                        "ADMINISTRADOR",
                        "TESORERIA"
                )

                // Reportes operativos e inteligencia
                .requestMatchers(
                        "/reportes/**",
                        "/inteligencia/**"
                ).hasAnyRole(
                        "ADMINISTRADOR",
                        "OPERADOR"
                )

                // Administración de cuentas y roles
                .requestMatchers(
                        "/usuarios/**",
                        "/roles/**"
                ).hasRole(
                        "ADMINISTRADOR"
                )

                // Cualquier otra ruta exige autenticación
                .anyRequest()
                .authenticated()
        );

        http.formLogin(form -> form
                .loginPage("/login")
                .loginProcessingUrl("/login")
                .defaultSuccessUrl("/dashboard", true)
                .failureUrl("/login?error")
                .permitAll()
        );

        http.logout(logout -> logout
                .logoutUrl("/logout")
                .logoutSuccessUrl("/login?logout")
                .invalidateHttpSession(true)
                .clearAuthentication(true)
                .deleteCookies("VICHAYAL_SESSION")
                .permitAll()
        );

        http.exceptionHandling(exception -> exception
                .accessDeniedPage("/error/403")
        );

        return http.build();
    }

    @Bean
    public PasswordEncoder passwordEncoder() {
        return PasswordEncoderFactories
                .createDelegatingPasswordEncoder();
    }
}

package pe.vichaval.gestionriego.tesoreria.entity;

import jakarta.persistence.*;

import java.math.*;
import java.time.*;

import pe.vichaval.gestionriego.agricultor.entity.Agricultor;
import pe.vichaval.gestionriego.tesoreria.dto.PagoForm;

@Entity
@Table(name = "pagos")
public class Pago {
    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;
    @Column(nullable = false, unique = true, length = 30)
    private String codigo;
    @Column(nullable = false)
    private BigDecimal monto;
    @Column(nullable = false)
    private LocalDate fecha;
    @Column(nullable = false, length = 150)
    private String concepto;
    @Column(nullable = false, unique = true, length = 50)
    private String comprobante;
    @Column(nullable = false, length = 30)
    private String metodo;
    @Column(nullable = false, length = 30)
    private String estado;
    @Column(name = "motivo_rechazo", length = 500)
    private String motivoRechazo;
    @Column(name = "motivo_anulacion", length = 500)
    private String motivoAnulacion;
    @Column(name = "fecha_creacion", nullable = false)
    private OffsetDateTime fechaCreacion = OffsetDateTime.now();
    @Column(name = "fecha_actualizacion", nullable = false)
    private OffsetDateTime fechaActualizacion = OffsetDateTime.now();
    @ManyToOne(fetch = FetchType.LAZY, optional = false)
    @JoinColumn(name = "agricultor_id", nullable = false)
    private Agricultor agricultor;

    protected Pago() {
    }

    public Pago(String c, PagoForm f, Agricultor a) {
        codigo = c;
        estado = "EFECTIVO".equals(f.getMetodo()) ? "CONFIRMADO" : "PENDIENTE";
        cargar(f, a);
    }

    private void cargar(PagoForm f, Agricultor a) {
        monto = f.getMonto();
        fecha = f.getFecha();
        concepto = f.getConcepto();
        comprobante = f.getComprobante();
        metodo = f.getMetodo();
        agricultor = a;
        fechaActualizacion = OffsetDateTime.now();
    }

    public void confirmar() {
        if (!"PENDIENTE".equals(estado)) throw new IllegalStateException("Solo pagos pendientes");
        estado = "CONFIRMADO";
    }

    public void rechazar(String m) {
        if (!"PENDIENTE".equals(estado)) throw new IllegalStateException("Solo pagos pendientes");
        if (m == null || m.isBlank()) throw new IllegalArgumentException("Motivo obligatorio");
        estado = "RECHAZADO";
        motivoRechazo = m;
    }

    public void anular(String m) {
        if (!"CONFIRMADO".equals(estado)) throw new IllegalStateException("Solo pagos confirmados");
        if (m == null || m.isBlank()) throw new IllegalArgumentException("Motivo obligatorio");
        estado = "ANULADO";
        motivoAnulacion = m;
    }

    public Long getId() {
        return id;
    }

    public String getCodigo() {
        return codigo;
    }

    public BigDecimal getMonto() {
        return monto;
    }

    public LocalDate getFecha() {
        return fecha;
    }

    public String getConcepto() {
        return concepto;
    }

    public String getComprobante() {
        return comprobante;
    }

    public String getMetodo() {
        return metodo;
    }

    public String getEstado() {
        return estado;
    }

    public Agricultor getAgricultor() {
        return agricultor;
    }
}

package pe.vichaval.gestionriego.finanzas.entity;

import jakarta.persistence.*;

import java.math.BigDecimal;
import java.time.*;

import pe.vichaval.gestionriego.agricultor.entity.Agricultor;
import pe.vichaval.gestionriego.finanzas.dto.CuentaCobrarForm;

@Entity
@Table(name = "cuentas_cobrar")
public class CuentaCobrar {
    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;
    @Column(nullable = false, unique = true, length = 30)
    private String codigo;
    @ManyToOne(fetch = FetchType.LAZY, optional = false)
    @JoinColumn(name = "agricultor_id")
    private Agricultor agricultor;
    @ManyToOne(fetch = FetchType.LAZY, optional = false)
    @JoinColumn(name = "concepto_id")
    private ConceptoCobro concepto;
    @Column(name = "monto_total", nullable = false)
    private BigDecimal montoTotal;
    @Column(name = "monto_pagado", nullable = false)
    private BigDecimal montoPagado = BigDecimal.ZERO;
    @Column(name = "fecha_emision", nullable = false)
    private LocalDate fechaEmision;
    @Column(name = "fecha_vencimiento", nullable = false)
    private LocalDate fechaVencimiento;
    @Column(nullable = false, length = 30)
    private String estado = "PENDIENTE";
    @Column(length = 500)
    private String observaciones;
    @Column(name = "motivo_anulacion", length = 500)
    private String motivoAnulacion;

    protected CuentaCobrar() {
    }

    public CuentaCobrar(String codigo, CuentaCobrarForm f, Agricultor a, ConceptoCobro c) {
        this.codigo = codigo;
        agricultor = a;
        concepto = c;
        montoTotal = f.getMontoTotal();
        fechaEmision = f.getFechaEmision();
        fechaVencimiento = f.getFechaVencimiento();
        observaciones = f.getObservaciones();
    }

    public BigDecimal getSaldoPendiente() {
        return montoTotal.subtract(montoPagado);
    }

    public void aplicarPago(BigDecimal monto) {
        if (monto.compareTo(getSaldoPendiente()) > 0)
            throw new IllegalStateException("El pago supera el saldo pendiente");
        montoPagado = montoPagado.add(monto);
        recalcular();
    }

    public void revertirPago(BigDecimal monto) {
        montoPagado = montoPagado.subtract(monto);
        if (montoPagado.signum() < 0) montoPagado = BigDecimal.ZERO;
        recalcular();
    }

    public void actualizarVencida() {
        if (("PENDIENTE".equals(estado) || "PAGO_PARCIAL".equals(estado)) && fechaVencimiento.isBefore(LocalDate.now()))
            estado = "VENCIDA";
    }

    public void anular(String motivo) {
        if ("PAGADA".equals(estado)) throw new IllegalStateException("No se puede anular una cuenta pagada");
        if (motivo == null || motivo.isBlank()) throw new IllegalArgumentException("Motivo obligatorio");
        estado = "ANULADA";
        motivoAnulacion = motivo;
    }

    private void recalcular() {
        if (getSaldoPendiente().signum() == 0) estado = "PAGADA";
        else if (montoPagado.signum() > 0)
            estado = fechaVencimiento.isBefore(LocalDate.now()) ? "VENCIDA" : "PAGO_PARCIAL";
        else estado = fechaVencimiento.isBefore(LocalDate.now()) ? "VENCIDA" : "PENDIENTE";
    }

    public Long getId() {
        return id;
    }

    public String getCodigo() {
        return codigo;
    }

    public Agricultor getAgricultor() {
        return agricultor;
    }

    public ConceptoCobro getConcepto() {
        return concepto;
    }

    public BigDecimal getMontoTotal() {
        return montoTotal;
    }

    public BigDecimal getMontoPagado() {
        return montoPagado;
    }

    public LocalDate getFechaEmision() {
        return fechaEmision;
    }

    public LocalDate getFechaVencimiento() {
        return fechaVencimiento;
    }

    public String getEstado() {
        return estado;
    }

    public String getObservaciones() {
        return observaciones;
    }
}

    }

import org.springframework.stereotype.Service;

import pe.vichaval.gestionriego.tesoreria.entity.Pago;
import pe.vichaval.gestionriego.tesoreria.repository.PagoRepository;

@Service
@Transactional
public class PagoService {
    private final PagoRepository p;
    private final AgricultorRepository a;

    public PagoService(PagoRepository p, AgricultorRepository a) {
        this.p = p;
        this.a = a;
    }

    public Pago get(Long id) {
        return p.findById(id).orElseThrow(() -> new IllegalArgumentException("Pago no encontrado"));
    }

    public void registrar(PagoForm f) {
        if (p.existsByComprobante(f.getComprobante())) throw new IllegalArgumentException("Comprobante duplicado");
        var ag = a.findById(f.getAgricultorId()).orElseThrow();
        p.save(new Pago("PG-%06d".formatted(p.count() + 1), f, ag));
    }

    public void confirmar(Long id) {
        get(id).confirmar();
    }

    public void rechazar(Long id, String m) {
        get(id).rechazar(m);
    }

    public void anular(Long id, String m) {
        get(id).anular(m);
    }
}

package pe.vichaval.gestionriego.finanzas.service;

import java.util.List;

import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import pe.vichaval.gestionriego.agricultor.repository.AgricultorRepository;
import pe.vichaval.gestionriego.finanzas.dto.ConceptoCobroForm;
import pe.vichaval.gestionriego.finanzas.dto.CuentaCobrarForm;
import pe.vichaval.gestionriego.finanzas.dto.PagoCuentaForm;
import pe.vichaval.gestionriego.finanzas.entity.ConceptoCobro;
import pe.vichaval.gestionriego.finanzas.entity.CuentaCobrar;
import pe.vichaval.gestionriego.finanzas.repository.ConceptoCobroRepository;
import pe.vichaval.gestionriego.finanzas.repository.CuentaCobrarRepository;
import pe.vichaval.gestionriego.tesoreria.dto.PagoForm;
import pe.vichaval.gestionriego.tesoreria.service.PagoService;

@Service
@Transactional
public class FinanzasService {

    private final CuentaCobrarRepository cuentas;
    private final ConceptoCobroRepository conceptos;
    private final AgricultorRepository agricultores;
    private final PagoService pagos;

    public FinanzasService(
            CuentaCobrarRepository cuentas,
            ConceptoCobroRepository conceptos,
            AgricultorRepository agricultores,
            PagoService pagos
    ) {
        this.cuentas = cuentas;
        this.conceptos = conceptos;
        this.agricultores = agricultores;
        this.pagos = pagos;
    }

    public List<CuentaCobrar> listar() {

        var lista = cuentas.findAll();

        lista.forEach(
            CuentaCobrar::actualizarVencida
        );

        return lista;
    }

    @Transactional(readOnly = true)
    public CuentaCobrar get(Long id) {

        return cuentas
            .findById(id)
            .orElseThrow(() ->
                new IllegalArgumentException(
                    "Cuenta no encontrada."
                )
            );
    }

    public void crearCuenta(
            CuentaCobrarForm form
    ) {
        if (
            form.getFechaEmision() == null
            || form.getFechaVencimiento() == null
        ) {
            throw new IllegalArgumentException(
                "Las fechas de emisión y vencimiento son obligatorias."
            );
        }

        if (
            form.getFechaVencimiento()
                .isBefore(form.getFechaEmision())
        ) {
            throw new IllegalArgumentException(
                "El vencimiento no puede ser anterior a la emisión."
            );
        }

        var agricultor = agricultores
            .findById(form.getAgricultorId())
            .orElseThrow(() ->
                new IllegalArgumentException(
                    "El agricultor seleccionado no existe."
                )
            );

        if (!"ACTIVO".equals(agricultor.getEstado())) {
            throw new IllegalStateException(
                "No se puede emitir la cuenta a un agricultor inactivo."
            );
        }

        var concepto = conceptos
            .findById(form.getConceptoId())
            .orElseThrow(() ->
                new IllegalArgumentException(
                    "El concepto seleccionado no existe."
                )
            );

        if (!concepto.isActivo()) {
            throw new IllegalStateException(
                "El concepto seleccionado está inactivo."
            );
        }

        String codigo = "CC-%06d".formatted(
            cuentas.count() + 1
        );

        CuentaCobrar cuenta = new CuentaCobrar(
            codigo,
            form,
            agricultor,
            concepto
        );

        cuentas.save(cuenta);
    }

    public void registrarPago(
            Long cuentaId,
            PagoCuentaForm form
    ) {
        CuentaCobrar cuenta = get(cuentaId);

        if (
            "ANULADA".equals(cuenta.getEstado())
            || "PAGADA".equals(cuenta.getEstado())
        ) {
            throw new IllegalStateException(
                "La cuenta no admite pagos."
            );
        }

        if (
            form.getMonto() == null
            || form.getMonto().signum() <= 0
        ) {
            throw new IllegalArgumentException(
                "El monto debe ser mayor que cero."
            );
        }

        if (
            form.getMonto()
                .compareTo(cuenta.getSaldoPendiente()) > 0
        ) {
            throw new IllegalStateException(
                "El pago supera el saldo pendiente."
            );
        }

        /*
         * Primero se aplica el pago a la cuenta.
         * Si PagoService falla, la transacción completa
         * se revierte automáticamente.
         */
        cuenta.aplicarPago(
            form.getMonto()
        );

        PagoForm pagoForm = new PagoForm();

        pagoForm.setAgricultorId(
            cuenta.getAgricultor().getId()
        );

        pagoForm.setMonto(
            form.getMonto()
        );

        pagoForm.setFecha(
            form.getFecha()
        );

        pagoForm.setConcepto(
            cuenta.getConcepto().getNombre()
            + " - "
            + cuenta.getCodigo()
        );

        pagoForm.setComprobante(
            form.getComprobante()
        );

        pagoForm.setMetodo(
            form.getMetodo()
        );

        pagos.registrar(
            pagoForm
        );

        cuentas.save(cuenta);
    }

    public void anular(
            Long id,
            String motivo
    ) {
        CuentaCobrar cuenta = get(id);

        cuenta.anular(motivo);

        cuentas.save(cuenta);
    }

    @Transactional(readOnly = true)
    public List<ConceptoCobro> listarConceptos() {

        return conceptos
            .findAllByOrderByNombre();
    }

    @Transactional(readOnly = true)
    public List<ConceptoCobro> conceptosActivos() {

        return conceptos
            .findByActivoTrueOrderByNombre();
    }

    public void crearConcepto(
            ConceptoCobroForm form
    ) {
        if (form == null) {
            throw new IllegalArgumentException(
                "Los datos del concepto son obligatorios."
            );
        }

        /*
         * Convierte, por ejemplo:
         *
         * derecho  -> DERECHO
         * Derecho  -> DERECHO
         */
        String codigo = form.getCodigo() == null
                ? ""
                : form.getCodigo()
                    .trim()
                    .toUpperCase();

        String nombre = form.getNombre() == null
                ? ""
                : form.getNombre().trim();

        String descripcion =
            form.getDescripcion() == null
                ? null
                : form.getDescripcion().trim();

        if (codigo.isBlank()) {
            throw new IllegalArgumentException(
                "El código es obligatorio."
            );
        }

        if (nombre.isBlank()) {
            throw new IllegalArgumentException(
                "El nombre es obligatorio."
            );
        }

        if (
            conceptos.existsByCodigoIgnoreCase(
                codigo
            )
        ) {
            throw new IllegalArgumentException(
                "El código "
                + codigo
                + " ya está registrado."
            );
        }

        /*
         * Se guardan los valores normalizados en el formulario.
         */
        form.setCodigo(codigo);
        form.setNombre(nombre);

        if (
            descripcion == null
            || descripcion.isBlank()
        ) {
            form.setDescripcion(null);
        } else {
            form.setDescripcion(descripcion);
        }

        ConceptoCobro concepto =
            new ConceptoCobro(form);

        conceptos.save(concepto);
    }
}
