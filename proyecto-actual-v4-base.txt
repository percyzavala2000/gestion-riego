package pe.vichaval.gestionriego.riego.entity;

import jakarta.persistence.*;

import java.time.*;

import pe.vichaval.gestionriego.parcela.entity.Parcela;

@Entity
@Table(name = "turnos_riego")
public class TurnoRiego {
    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;
    @Column(nullable = false, unique = true)
    private String codigo;
    @Column(nullable = false)
    private LocalDate fecha;
    @Column(name = "hora_inicio", nullable = false)
    private LocalTime horaInicio;
    @Column(name = "hora_fin", nullable = false)
    private LocalTime horaFin;
    @Column(nullable = false)
    private String estado = "PROGRAMADO";
    private String observaciones;
    @ManyToOne(fetch = FetchType.LAZY,optional = false)
    @JoinColumn(name = "parcela_id", nullable = false   )
    private Parcela parcela;

    protected TurnoRiego() {
    }

    public TurnoRiego(String c, LocalDate f, LocalTime i, LocalTime fin, String o, Parcela p) {
        codigo = c;
        fecha = f;
        horaInicio = i;
        horaFin = fin;
        observaciones = o;
        parcela = p;
    }

    public String getCodigo() {
        return codigo;
    }

    public LocalDate getFecha() {
        return fecha;
    }

    public LocalTime getHoraInicio() {
        return horaInicio;
    }

    public LocalTime getHoraFin() {
        return horaFin;
    }

    public String getEstado() {
        return estado;
    }

    public Parcela getParcela() {
        return parcela;
    }
}

package pe.vichaval.gestionriego.agricultor.controller;

import jakarta.validation.Valid;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.validation.BindingResult;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.ModelAttribute;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.servlet.mvc.support.RedirectAttributes;
import pe.vichaval.gestionriego.agricultor.dto.AgricultorForm;
import pe.vichaval.gestionriego.agricultor.service.AgricultorService;

@Controller
@RequestMapping("/agricultores")
public class AgricultorWebController {

    private final AgricultorService agricultorService;

    public AgricultorWebController(
            AgricultorService agricultorService
    ) {
        this.agricultorService = agricultorService;
    }

    @GetMapping
    public String listar(
            @RequestParam(defaultValue = "") String q,
            @RequestParam(defaultValue = "0") int page,
            Model model
    ) {
        model.addAttribute(
                "pagina",
                agricultorService.listar(q, page)
        );

        model.addAttribute("q", q);

        return "agricultor/list";
    }

    @GetMapping("/nuevo")
    public String mostrarFormularioNuevo(Model model) {

        model.addAttribute(
                "form",
                new AgricultorForm()
        );

        model.addAttribute("id", null);

        return "agricultor/form";
    }

    @GetMapping("/{id}/editar")
    public String mostrarFormularioEditar(
            @PathVariable Long id,
            Model model
    ) {
        var agricultor = agricultorService.get(id);

        AgricultorForm form = new AgricultorForm();

        form.setDni(agricultor.getDni());
        form.setNombres(agricultor.getNombres());
        form.setApellidos(agricultor.getApellidos());
        form.setTelefono(agricultor.getTelefono());
        form.setSector(agricultor.getSector());
        form.setEstado(agricultor.getEstado());

        model.addAttribute("form", form);
        model.addAttribute("id", id);

        return "agricultor/form";
    }

    @PostMapping("/guardar")
    public String guardar(
            @RequestParam(required = false) Long id,
            @Valid @ModelAttribute("form") AgricultorForm form,
            BindingResult bindingResult,
            Model model,
            RedirectAttributes redirectAttributes
    ) {
        model.addAttribute("id", id);

        if (bindingResult.hasErrors()) {
            return "agricultor/form";
        }

        try {
            agricultorService.save(form, id);

            if (id == null) {
                redirectAttributes.addFlashAttribute(
                        "mensajeExito",
                        "Agricultor registrado correctamente."
                );
            } else {
                redirectAttributes.addFlashAttribute(
                        "mensajeExito",
                        "Agricultor actualizado correctamente."
                );
            }

            return "redirect:/agricultores";

        } catch (IllegalArgumentException exception) {

            bindingResult.reject(
                    "errorNegocio",
                    exception.getMessage()
            );

            return "agricultor/form";
        }
    }
}

package pe.vichaval.gestionriego.config;

import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.security.config.annotation.method.configuration.EnableMethodSecurity;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.crypto.factory.PasswordEncoderFactories;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.security.web.SecurityFilterChain;

@Configuration
@EnableMethodSecurity
public class SecurityConfig {

    @Bean
    public SecurityFilterChain securityFilterChain(
            HttpSecurity http
    ) throws Exception {

        http.authorizeHttpRequests(authorize -> authorize

                // Recursos públicos
                .requestMatchers(
                        "/login",
                        "/error",
                        "/error/**",
                        "/css/**",
                        "/js/**",
                        "/images/**",
                        "/icons/**",
                        "/favicon.ico",
                        "/actuator/health"
                ).permitAll()

                // Acceso compartido al dashboard
                .requestMatchers(
                        "/",
                        "/dashboard"
                ).hasAnyRole(
                        "ADMINISTRADOR",
                        "OPERADOR",
                        "TESORERIA"
                )

                // Gestión operativa
                .requestMatchers(
                        "/agricultores/**",
                        "/parcelas/**",
                        "/turnos/**"
                ).hasAnyRole(
                        "ADMINISTRADOR",
                        "OPERADOR"
                )

                // Gestión económica
                .requestMatchers(
                        "/tesoreria/**"
                ).hasAnyRole(
                        "ADMINISTRADOR",
                        "TESORERIA"
                )

                // Reportes financieros
                .requestMatchers(
                        "/reportes/financieros/**"
                ).hasAnyRole(
                        "ADMINISTRADOR",
                        "TESORERIA"
                )

                // Reportes operativos e inteligencia
                .requestMatchers(
                        "/reportes/**",
                        "/inteligencia/**"
                ).hasAnyRole(
                        "ADMINISTRADOR",
                        "OPERADOR"
                )

                // Administración de cuentas y roles
                .requestMatchers(
                        "/usuarios/**",
                        "/roles/**"
                ).hasRole(
                        "ADMINISTRADOR"
                )

                // Cualquier otra ruta exige autenticación
                .anyRequest()
                .authenticated()
        );

        http.formLogin(form -> form
                .loginPage("/login")
                .loginProcessingUrl("/login")
                .defaultSuccessUrl("/dashboard", true)
                .failureUrl("/login?error")
                .permitAll()
        );

        http.logout(logout -> logout
                .logoutUrl("/logout")
                .logoutSuccessUrl("/login?logout")
                .invalidateHttpSession(true)
                .clearAuthentication(true)
                .deleteCookies("VICHAYAL_SESSION")
                .permitAll()
        );

        http.exceptionHandling(exception -> exception
                .accessDeniedPage("/error/403")
        );

        return http.build();
    }

    @Bean
    public PasswordEncoder passwordEncoder() {
        return PasswordEncoderFactories
                .createDelegatingPasswordEncoder();
    }
}

package pe.vichaval.gestionriego.security.service;

import java.util.Collection;
import java.util.List;

import org.springframework.security.core.GrantedAuthority;
import org.springframework.security.core.authority.SimpleGrantedAuthority;
import org.springframework.security.core.userdetails.UserDetails;

import pe.vichaval.gestionriego.usuario.entity.Usuario;

public class UsuarioPrincipal implements UserDetails {

    private final Long id;
    private final String username;
    private final String password;
    private final String nombres;
    private final String apellidos;
    private final String rol;
    private final boolean activo;

    public UsuarioPrincipal(Usuario usuario) {
        this.id = usuario.getId();
        this.username = usuario.getUsername();
        this.password = usuario.getPasswordHash();
        this.nombres = usuario.getNombres();
        this.apellidos = usuario.getApellidos();
        this.rol = usuario.getRol().getCodigo().name();
        this.activo = usuario.isActivo();
    }

    public Long getId() {
        return id;
    }

    public String getNombres() {
        return nombres;
    }

    public String getApellidos() {
        return apellidos;
    }

    public String getNombreCompleto() {
        return nombres + " " + apellidos;
    }

    public String getRol() {
        return rol;
    }

    public String getRolLegible() {
        return switch (rol) {
            case "ADMINISTRADOR" -> "Administrador";
            case "TESORERIA" -> "Tesorería";
            case "OPERADOR" -> "Operador de riego";
            default -> rol;
        };
    }

    public String getIniciales() {
        String inicialNombre = nombres == null || nombres.isBlank()
                ? ""
                : nombres.substring(0, 1).toUpperCase();

        String inicialApellido = apellidos == null || apellidos.isBlank()
                ? ""
                : apellidos.substring(0, 1).toUpperCase();

        return inicialNombre + inicialApellido;
    }

    @Override
    public Collection<? extends GrantedAuthority> getAuthorities() {
        return List.of(
                new SimpleGrantedAuthority("ROLE_" + rol)
        );
    }

    @Override
    public String getPassword() {
        return password;
    }

    @Override
    public String getUsername() {
        return username;
    }

    @Override
    public boolean isAccountNonExpired() {
        return activo;
    }

    @Override
    public boolean isAccountNonLocked() {
        return activo;
    }

    @Override
    public boolean isCredentialsNonExpired() {
        return activo;
    }

    @Override
    public boolean isEnabled() {
        return activo;
    }
}

import org.springframework.security.core.userdetails.UserDetailsService;

import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import pe.vichaval.gestionriego.usuario.entity.Usuario;
import pe.vichaval.gestionriego.usuario.repository.UsuarioRepository;

@Service
public class CustomUserDetailsService implements UserDetailsService {

    private final UsuarioRepository usuarioRepository;

    public CustomUserDetailsService(
            UsuarioRepository usuarioRepository
    ) {
        this.usuarioRepository = usuarioRepository;
    }

    @Override
    @Transactional(readOnly = true)
    public UserDetails loadUserByUsername(String username)
            throws UsernameNotFoundException {

        Usuario usuario = usuarioRepository
                .findByUsernameIgnoreCase(username)
                .orElseThrow(() ->
                        new UsernameNotFoundException(
                                "Usuario o contraseña incorrectos"
                        )
                );

        return new UsuarioPrincipal(usuario);
    }
}

package pe.vichaval.gestionriego.dashboard.service;

import java.math.BigDecimal;
import java.time.LocalDate;

import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import pe.vichaval.gestionriego.agricultor.repository.AgricultorRepository;
import pe.vichaval.gestionriego.dashboard.dto.DashboardResumen;
import pe.vichaval.gestionriego.parcela.repository.ParcelaRepository;
import pe.vichaval.gestionriego.riego.repository.TurnoRiegoRepository;
import pe.vichaval.gestionriego.tesoreria.repository.PagoRepository;

@Service
public class DashboardService {

    private final AgricultorRepository agricultorRepository;
    private final ParcelaRepository parcelaRepository;
    private final TurnoRiegoRepository turnoRepository;
    private final PagoRepository pagoRepository;

    public DashboardService(
            AgricultorRepository agricultorRepository,
            ParcelaRepository parcelaRepository,
            TurnoRiegoRepository turnoRepository,
            PagoRepository pagoRepository
    ) {
        this.agricultorRepository = agricultorRepository;
        this.parcelaRepository = parcelaRepository;
        this.turnoRepository = turnoRepository;
        this.pagoRepository = pagoRepository;
    }

    @Transactional(readOnly = true)
    public DashboardResumen obtenerResumen() {

        LocalDate fechaActual = LocalDate.now();

        long turnosHoy =
                turnoRepository.countByFecha(fechaActual);

        long turnosProgramados =
                turnoRepository.countByFechaAndEstado(
                        fechaActual,
                        "PROGRAMADO"
                );

        long agricultoresActivos =
                agricultorRepository.countByEstado("ACTIVO");

        long parcelasRegistradas =
                parcelaRepository.count();

        BigDecimal totalIngresos =
                pagoRepository.total();

        long cantidadPagos =
                pagoRepository.count();

        /*
         * Estos valores serán sustituidos cuando se implemente
         * la entidad CuentaPorCobrar o Deuda.
         */
        BigDecimal totalDeuda =
                BigDecimal.ZERO;

        long agricultoresConDeuda =
                0L;

        return new DashboardResumen(
                turnosHoy,
                turnosProgramados,
                agricultoresActivos,
                parcelasRegistradas,
                totalIngresos,
                cantidadPagos,
                totalDeuda,
                agricultoresConDeuda
        );
    }
}

package pe.vichaval.gestionriego.usuario.entity;

import jakarta.persistence.*;

import java.time.OffsetDateTime;

@Entity
@Table(name = "usuarios")
public class Usuario {
    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;
    @Column(nullable = false, unique = true, length = 50)
    private String username;
    @Column(name = "password_hash", nullable = false, length = 255)
    private String passwordHash;
    @Column(nullable = false, length = 100)
    private String nombres;
    @Column(nullable = false, length = 120)
    private String apellidos;
    @Column(nullable = false)
    private boolean activo = true;
    @ManyToOne(fetch = FetchType.EAGER, optional = false)
    @JoinColumn(name = "rol_id", nullable = false)
    private Rol rol;
    @Column(name = "fecha_creacion", nullable = false)
    private OffsetDateTime fechaCreacion;

    protected Usuario() {
    }

    public Usuario(String u, String p, String n, String a, Rol r) {
        username = u;
        passwordHash = p;
        nombres = n;
        apellidos = a;

CREATE TABLE parcelas(id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,codigo VARCHAR(30) NOT NULL UNIQUE,nombre VARCHAR(100) NOT NULL,hectareas NUMERIC(10,2) NOT NULL,cultivo VARCHAR(100),agricultor_id BIGINT NOT NULL REFERENCES agricultores(id));

CREATE TABLE pagos(id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,monto NUMERIC(12,2) NOT NULL,fecha DATE NOT NULL,concepto VARCHAR(150) NOT NULL,comprobante VARCHAR(50) NOT NULL UNIQUE,metodo VARCHAR(30) NOT NULL,agricultor_id BIGINT NOT NULL REFERENCES agricultores(id));
